IMUL:          MOV E,B      //   E = B       // save parameter

LOOP:          ADD D        //     A = A+D
               DCR E        //     E = E-1

MAIN:          LXI SP,00A6  //   SP = 00A6h // stack
               LXI H,0050   //   HL = 0050h  // datum

               MVI D,01     //   D  = 1

FAT:           CALL IMUL    //     D = D*B
